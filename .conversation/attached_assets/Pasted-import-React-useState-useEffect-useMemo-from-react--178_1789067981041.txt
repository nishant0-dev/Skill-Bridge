                  onClick={handleApproveDelta}
                  className={`flex items-center justify-center gap-2 px-5 py-2.5 rounded-xl text-xs sm:text-sm font-semibold transition-all shadow-lg active:scale-95 ${
                    approvedSchemes[currentScheme.id]
                      ? "bg-emerald-600 hover:bg-emerald-500 text-white shadow-emerald-600/30"
                      : "bg-gradient-to-r from-indigo-600 to-cyan-600 hover:from-indigo-500 hover:to-cyan-500 text-white shadow-indigo-600/30"
                  }`}
                >
                  {approvedSchemes[currentScheme.id] ? (
                    <>
                      <Icon name="checkCircle" size={16} />
                      <span>Delta Approved · Export Ready</span>
                    </>
                  ) : (
                    <>
                      <Icon name="send" size={16} />
                      <span>Approve Delta & Export Syllabus</span>
                    </>
                  )}
                </button>
              </div>
            </div>

            {/* Metrics Ribbon */}
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
              <div className="p-4 rounded-xl bg-slate-900/80 border border-slate-800">
                <span className="text-xs text-slate-400">Current Syllabus Alignment</span>
                <div className="flex items-center gap-2 mt-1">
                  <span className="text-2xl font-black text-amber-400">{currentScheme.currentAlignment}%</span>
                  <span className="text-xs text-amber-300 bg-amber-500/10 px-2 py-0.5 rounded border border-amber-500/20">
                    Legacy Gaps
                  </span>
                </div>
              </div>

              <div className="p-4 rounded-xl bg-slate-900/80 border border-slate-800">
                <span className="text-xs text-slate-400">Projected Alignment (Post-Delta)</span>
                <div className="flex items-center gap-2 mt-1">
                  <span className="text-2xl font-black text-emerald-400">{currentScheme.projectedAlignment}%</span>
                  <span className="text-xs text-emerald-300 bg-emerald-500/10 px-2 py-0.5 rounded border border-emerald-500/20">
                    +{(currentScheme.projectedAlignment - currentScheme.currentAlignment)}% Parity
                  </span>
                </div>
              </div>

              <div className="p-4 rounded-xl bg-slate-900/80 border border-slate-800">
                <span className="text-xs text-slate-400">Credit Hours Reclaimed</span>
                <div className="flex items-center gap-2 mt-1">
                  <span className="text-2xl font-black text-indigo-400">{currentScheme.reclaimedHours} Hours</span>
                  <Icon name="clock" size={16} className="text-indigo-400" />
                </div>
              </div>
            </div>

            {/* SIDE-BY-SIDE COMPARISON TABLE */}
            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* LEFT: Outdated Topics Flagged for Removal */}
              <div className="rounded-2xl bg-slate-900/90 border border-rose-900/30 overflow-hidden shadow-2xl">
                <div className="px-5 py-4 bg-gradient-to-r from-rose-950/60 to-slate-900 border-b border-rose-900/40 flex items-center justify-between">
                  <div className="flex items-center gap-2.5">
                    <div className="p-1.5 rounded-lg bg-rose-500/20 text-rose-400">
                      <Icon name="trash" size={16} />
                    </div>
                    <div>
                      <h4 className="text-sm font-bold text-white">Outdated Topics Flagged for Removal</h4>
                      <p className="text-[11px] text-rose-300/80">Obsolescence detected based on live hiring signals</p>
                    </div>
                  </div>
                  <span className="text-xs font-mono font-bold text-rose-400 bg-rose-500/10 px-2.5 py-1 rounded-full border border-rose-500/20">
                    {currentScheme.outdatedTopics.length} Flagged
                  </span>
                </div>

                <div className="p-3 space-y-3">
                  {currentScheme.outdatedTopics.map((topic) => (
                    <div key={topic.code} className="p-4 rounded-xl bg-slate-800/40 border border-slate-800">
                      <div className="flex items-start justify-between gap-3">
                        <div>
                          <div className="flex items-center gap-2">
                            <span className="px-1.5 py-0.5 rounded font-mono text-[10px] font-bold bg-slate-700 text-slate-300">
                              {topic.code}
                            </span>
                            <h5 className="text-xs sm:text-sm font-semibold text-rose-200">{topic.title}</h5>
                          </div>
                          <p className="mt-1.5 text-xs text-slate-400 leading-relaxed">{topic.reason}</p>
                        </div>
                        <span className="px-2 py-0.5 rounded text-[10px] font-bold bg-rose-500/20 text-rose-300 border border-rose-500/30">
                          {topic.severity}
                        </span>
                      </div>
                      <div className="mt-3 pt-2.5 border-t border-slate-800/60 flex items-center justify-between text-[11px]">
                        <span className="text-slate-500">
                          Reclaims: <strong className="text-rose-300">{topic.hoursReclaimed} Lecture Hours</strong>
                        </span>
                        <span className="text-cyan-400 font-medium">Flagged for Replacement</span>
                      </div>
                    </div>
                  ))}
                </div>
              </div>

              {/* RIGHT: Industry Demanded Modules Recommended to Add */}
              <div className="rounded-2xl bg-slate-900/90 border border-emerald-900/30 overflow-hidden shadow-2xl">
                <div className="px-5 py-4 bg-gradient-to-r from-emerald-950/60 to-slate-900 border-b border-emerald-900/40 flex items-center justify-between">
                  <div className="flex items-center gap-2.5">
                    <div className="p-1.5 rounded-lg bg-emerald-500/20 text-emerald-400">
                      <Icon name="plusCircle" size={16} />
                    </div>
                    <div>
                      <h4 className="text-sm font-bold text-white">Industry Demanded Modules Recommended to Add</h4>
                      <p className="text-[11px] text-emerald-300/80">Direct mapping to high-growth vacancies in top tech districts</p>
                    </div>
                  </div>
                  <span className="text-xs font-mono font-bold text-emerald-400 bg-emerald-500/10 px-2.5 py-1 rounded-full border border-emerald-500/20">
                    {currentScheme.demandedModules.length} Modules
                  </span>
                </div>

                <div className="p-3 space-y-3">
                  {currentScheme.demandedModules.map((module) => (
                    <div key={module.code} className="p-4 rounded-xl bg-slate-800/40 border border-slate-800">
                      <div className="flex items-start justify-between gap-3">
                        <div>
                          <div className="flex items-center gap-2">
                            <span className="px-1.5 py-0.5 rounded font-mono text-[10px] font-bold bg-emerald-500/20 text-emerald-300 border border-emerald-500/30">
                              {module.code}
                            </span>
                            <h5 className="text-xs sm:text-sm font-semibold text-emerald-200">{module.title}</h5>
                          </div>
                          <div className="mt-2 flex flex-wrap gap-1.5">
                            {module.roles.map((role) => (
                              <span
                                key={role}
                                className="px-2 py-0.5 rounded text-[10px] font-medium bg-slate-700/60 text-slate-300 border border-slate-700"
                              >
                                {role}
                              </span>
                            ))}
                          </div>
                        </div>
                        <div className="text-right">
                          <span className="text-xs font-mono font-bold text-emerald-400 bg-emerald-500/10 px-2 py-0.5 rounded border border-emerald-500/20">
                            {module.industryDemand}% Demand
                          </span>
                          <p className="text-[10px] text-slate-400 mt-1">NSQF Lvl {module.nsqfLevel}</p>
                        </div>
                      </div>
                      <div className="mt-3 pt-2.5 border-t border-slate-800/60 flex items-center justify-between text-[11px]">
                        <span className="text-slate-500">
                          Allocated Lab: <strong className="text-emerald-300">{module.labHours} Practical Hours</strong>
                        </span>
                        <span className="text-indigo-300 font-semibold font-mono">NEP 2020 Compliant</span>
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* =============================================================== */}
        {/* VIEW 3: ADAPTIVE ASSESSMENT & READINESS (STUDENT VIEW)           */}
        {/* =============================================================== */}
        {activeTab === "assessment" && (
          <div className="space-y-6 animate-in fade-in duration-300">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 p-5 rounded-2xl bg-slate-900/90 border border-slate-800 shadow-xl">
              <div>
                <div className="flex items-center gap-2">
                  <Icon name="brain" size={18} className="text-indigo-400" />
                  <h3 className="text-base sm:text-lg font-bold text-white">
                    Adaptive Assessment & Student Readiness Engine
                  </h3>
                </div>
                <p className="text-xs text-slate-400 mt-1">
                  Dynamic 3-question adaptive quiz simulator shifting difficulty badge (Easy → Medium → Hard) in real time.
                </p>
              </div>
              <div className="flex items-center gap-2 text-xs text-slate-400 bg-slate-800/60 px-3 py-1.5 rounded-xl border border-slate-700">
                <Icon name="shield" size={16} className="text-emerald-400" />
                <span>NSQF Alignment Active</span>
              </div>
            </div>

            {!isQuizCompleted ? (
              /* ACTIVE 3-QUESTION ADAPTIVE SIMULATOR */
              <div className="max-w-3xl mx-auto space-y-6">
                {/* Stepper Bar & Real-time Difficulty Badge */}
                <div className="p-4 rounded-2xl bg-slate-900/80 border border-slate-800 shadow-lg space-y-3">
                  <div className="flex items-center justify-between">
                    <div className="flex items-center gap-3">
                      <span className="text-xs font-bold uppercase tracking-wider text-slate-400">
                        Question {currentQuizStep + 1} of 3
                      </span>
                      <span className="text-slate-600">|</span>
                      <span className="text-xs font-medium text-slate-300">
                        Domain: <strong className="text-indigo-300">{currentQuestion.domain}</strong>
                      </span>
                    </div>

                    {/* DYNAMIC DIFFICULTY BADGE (Easy -> Medium -> Hard) */}
                    <div>
                      {currentQuestion.difficulty === "Easy" && (
                        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-emerald-500/10 text-emerald-400 border border-emerald-500/30">
                          <span className="w-2 h-2 rounded-full bg-emerald-400"></span>
                          Difficulty: Easy
                        </span>
                      )}
                      {currentQuestion.difficulty === "Medium" && (
                        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-amber-500/10 text-amber-400 border border-amber-500/30">
                          <span className="w-2 h-2 rounded-full bg-amber-400"></span>
                          Difficulty: Medium ↗ (Adaptive Shift)
                        </span>
                      )}
                      {currentQuestion.difficulty === "Hard" && (
                        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-rose-500/10 text-rose-400 border border-rose-500/30">
                          <span className="w-2 h-2 rounded-full bg-rose-400 animate-pulse"></span>
                          Difficulty: Hard ↗↗ (Advanced Tier)
                        </span>
                      )}
                    </div>
                  </div>

                  <div className="grid grid-cols-3 gap-2">
                    {[0, 1, 2].map((idx) => (
                      <div
                        key={idx}
                        className={`h-2 rounded-full transition-all duration-300 ${
                          currentQuizStep > idx
                            ? "bg-emerald-500"
                            : currentQuizStep === idx
                            ? "bg-gradient-to-r from-indigo-500 to-cyan-500"
                            : "bg-slate-800"
                        }`}
                      ></div>
                    ))}
                  </div>
                </div>

                {/* Question Details Card */}
                <div className="p-6 sm:p-8 rounded-2xl bg-slate-900/90 border border-slate-800 shadow-2xl space-y-6">
                  <h4 className="text-base sm:text-lg font-semibold text-white leading-relaxed">
                    {currentQuestion.questionText}
                  </h4>

                  {currentQuestion.codeSnippet && (
                    <div className="rounded-xl bg-[#060a14] border border-slate-800 p-4 font-mono text-xs text-slate-200 overflow-x-auto">
                      <div className="text-[10px] text-slate-500 pb-2 mb-2 border-b border-slate-800 flex items-center justify-between">
                        <span>Industry Code Inspection</span>
                        <span>UTF-8</span>
                      </div>
                      <pre className="text-slate-300 leading-relaxed">
                        <code>{currentQuestion.codeSnippet}</code>
                      </pre>
                    </div>
                  )}

                  {/* Options */}
                  <div className="space-y-3">
                    {currentQuestion.options.map((opt) => {
                      const isSelected = selectedOptionId === opt.id;
                      let style = "bg-slate-800/50 border-slate-700/80 hover:bg-slate-800 text-slate-200";

                      if (isSelected && !isAnswerSubmitted) {
                        style = "bg-indigo-600/20 border-indigo-500 text-white ring-2 ring-indigo-500/30";
                      }

                      if (isAnswerSubmitted) {
                        if (opt.isCorrect) {
                          style = "bg-emerald-950/40 border-emerald-500 text-emerald-100 ring-2 ring-emerald-500/40";
                        } else if (isSelected && !opt.isCorrect) {
                          style = "bg-rose-950/40 border-rose-500 text-rose-200 ring-2 ring-rose-500/40";
                        } else {
                          style = "bg-slate-800/20 border-slate-800 text-slate-500 opacity-60";
                        }
                      }

                      return (
                        <div
                          key={opt.id}
                          onClick={() => handleSelectQuizOption(opt.id)}
                          className={`p-4 rounded-xl border text-xs sm:text-sm font-medium transition-all cursor-pointer flex items-start gap-3.5 ${style}`}
                        >
                          <div
                            className={`mt-0.5 w-5 h-5 rounded-full flex items-center justify-center text-xs font-bold flex-shrink-0 ${
                              isSelected && !isAnswerSubmitted
                                ? "bg-indigo-500 text-white"
                                : "bg-slate-700 text-slate-300"
                            }`}
                          >
                            {opt.id.toUpperCase()}
                          </div>
                          <div className="flex-1">
                            <p className="leading-relaxed">{opt.text}</p>
                            {isAnswerSubmitted && (isSelected || opt.isCorrect) && (
                              <p className={`mt-2 text-xs leading-relaxed ${opt.isCorrect ? "text-emerald-300" : "text-rose-300"}`}>
                                {opt.explanation}
                              </p>
                            )}
                          </div>
                        </div>
                      );
                    })}
                  </div>

                  {/* Actions */}
                  <div className="pt-4 border-t border-slate-800 flex flex-col sm:flex-row items-center justify-between gap-3">
                    <span className="text-xs text-slate-400">
                      {!isAnswerSubmitted
                        ? "Select an answer to inspect dynamic branching"
                        : "Adaptive engine recalculating difficulty..."}
                    </span>

                    {!isAnswerSubmitted ? (
                      <button
                        onClick={handleSubmitQuizAnswer}
                        disabled={!selectedOptionId}
                        className="w-full sm:w-auto px-6 py-2.5 rounded-xl bg-indigo-600 hover:bg-indigo-500 disabled:opacity-40 text-white font-semibold text-xs sm:text-sm shadow-lg shadow-indigo-600/30 transition-all active:scale-95"
                      >
                        Verify Answer
                      </button>
                    ) : (
                      <button
                        onClick={handleNextQuizQuestion}
                        className="w-full sm:w-auto flex items-center justify-center gap-2 px-6 py-2.5 rounded-xl bg-gradient-to-r from-emerald-600 to-indigo-600 hover:from-emerald-500 hover:to-indigo-500 text-white font-semibold text-xs sm:text-sm shadow-lg shadow-emerald-600/20 transition-all active:scale-95"
                      >
                        <span>{currentQuizStep < 2 ? "Next Adaptive Question" : "View Readiness Score"}</span>
                        <Icon name="arrowRight" size={16} />
                      </button>
                    )}
                  </div>
                </div>
              </div>
            ) : (
              /* FINAL RESULT CARD (Job Readiness Score: 82%, NSQF Level 5, 2-step remedial roadmap) */
              <div className="max-w-4xl mx-auto space-y-6 animate-in fade-in duration-500">
                <div className="p-6 sm:p-8 rounded-3xl bg-gradient-to-br from-slate-900 via-slate-900/90 to-indigo-950/60 border border-slate-800 shadow-2xl relative overflow-hidden">
                  <div className="flex flex-col md:flex-row items-center justify-between gap-8">
                    <div className="flex items-center gap-6">
                      <div className="relative flex items-center justify-center w-28 h-28 sm:w-32 sm:h-32 rounded-full bg-slate-950 border-4 border-indigo-500/30">
                        <div className="text-center">
                          <span className="text-3xl sm:text-4xl font-black text-white">82%</span>
                          <span className="block text-[10px] uppercase font-bold text-emerald-400">Readiness</span>
                        </div>
                      </div>

                      <div className="space-y-1.5">
                        <span className="px-2.5 py-0.5 rounded-full text-xs font-bold bg-emerald-500/10 text-emerald-400 border border-emerald-500/30">
                          Assessment Complete
                        </span>
                        <h4 className="text-xl sm:text-2xl font-black text-white">
                          Job Readiness Score: 82%
                        </h4>
                        <div className="flex items-center gap-2">
                          <span className="text-xs text-slate-300 font-semibold">NSQF Alignment Level:</span>
                          <span className="px-2.5 py-0.5 rounded-md text-xs font-bold bg-indigo-500/20 text-indigo-300 border border-indigo-500/30">
                            Level 5 (Industry Associate Engineer)
                          </span>
                        </div>
                        <p className="text-xs text-slate-400">
                          Evaluated by Blood Code IRT Assessment Engine against SIH26134 Industry Benchmark.
                        </p>
                      </div>
                    </div>

                    <div className="flex flex-col gap-3 w-full md:w-auto">
                      <button
                        onClick={handleResetQuiz}
                        className="flex items-center justify-center gap-2 px-5 py-2.5 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-700 text-slate-300 text-xs sm:text-sm font-medium transition-all active:scale-95"
                      >
                        <Icon name="rotateCcw" size={16} />
                        <span>Retake Adaptive Quiz</span>
                      </button>
                    </div>
                  </div>
                </div>

                {/* 2-STEP PERSONALIZED REMEDIAL ROADMAP */}
                <div className="p-6 sm:p-8 rounded-3xl bg-slate-900/90 border border-slate-800 shadow-2xl space-y-6">
                  <div>
                    <h4 className="text-base sm:text-lg font-bold text-white flex items-center gap-2">
                      <Icon name="trendingUp" size={18} className="text-indigo-400" />
                      <span>2-Step Personalized Remedial Roadmap</span>
                    </h4>
                    <p className="text-xs text-slate-400 mt-1">
                      Targeted micro-learning modules calibrated to elevate your Job Readiness from 82% to 98% (NSQF Level 6).
                    </p>
                  </div>

                  <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                    {REMEDIAL_ROADMAP_STEPS.map((step, idx) => (
                      <div key={step.title} className="p-6 rounded-2xl bg-slate-800/40 border border-slate-800 space-y-4">
                        <div className="flex items-center justify-between">
                          <span
                            className={`px-2.5 py-0.5 rounded-full text-[10px] font-bold ${
                              idx === 0
                                ? "bg-rose-500/20 text-rose-300 border border-rose-500/30"
                                : "bg-emerald-500/20 text-emerald-300 border border-emerald-500/30"
                            }`}
                          >
                            {step.badge}
                          </span>
                          <span className="text-xs font-mono text-slate-400">{step.duration}</span>
                        </div>
                        <h5 className="text-sm font-bold text-white">{step.title}</h5>
                        <p className="text-xs text-slate-400">{step.description}</p>
                        <ul className="space-y-1.5 text-xs text-slate-300">
                          {step.deliverables.map((item) => (
                            <li key={item} className="flex items-start gap-2">
                              <span className="text-emerald-400">✓</span>
                              <span>{item}</span>
                            </li>
                          ))}
                        </ul>
                      </div>
                    ))}
                  </div>
                </div>
              </div>
            )}
          </div>
        )}
      </main>

      {/* ---------------------------------------------------------------- */}
      {/* PROFESSIONAL FOOTER                                              */}
      {/* ---------------------------------------------------------------- */}
      <footer className="w-full border-t border-slate-800/80 bg-[#070b16] py-6 px-4 sm:px-8 mt-12 text-xs text-slate-400">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <div className="w-7 h-7 rounded-lg bg-indigo-600/30 border border-indigo-500/30 flex items-center justify-center text-indigo-400">
              <Icon name="layers" size={16} />
            </div>
            <div>
              <p className="font-semibold text-slate-200">
                Skill Bridge — AI-Powered Industry–Skill Alignment Platform
              </p>
              <p className="text-[11px] text-slate-500">
                Smart India Hackathon Prototype · Problem Statement <strong className="text-slate-400">SIH26134</strong>
              </p>
            </div>
          </div>
          <div className="flex flex-wrap items-center gap-4 text-[11px]">
            <span className="flex items-center gap-1 text-slate-400">
              <span>Engineered by</span>
              <strong className="text-white px-1.5 py-0.5 rounded bg-slate-800 border border-slate-700">Team Blood Code</strong>
            </span>
            <span className="text-slate-600">|</span>
            <span className="text-emerald-400 font-medium flex items-center gap-1">
              <Icon name="shield" size={14} />
              <span>NSQF & NEP 2020 Aligned</span>
            </span>
          </div>
        </div>
      </footer>
    </div>
  );
}
